Patch your current src/components/StreetViewer.jsx with this change.
This keeps your current transition/minimap design and only fixes the initial scene selection.

1) Add this helper near the top, after safeScenes(mapData):

function getSceneTitle(scene, fallback = "Untitled Location") {
  return scene?.label || scene?.title || scene?.name || scene?.id || fallback;
}

function getAlphabeticalFirstSceneId(mapData) {
  const scenes = Object.values(safeScenes(mapData)).filter(Boolean);

  return [...scenes].sort((a, b) =>
    getSceneTitle(a, a?.id).localeCompare(getSceneTitle(b, b?.id), undefined, {
      numeric: true,
      sensitivity: "base",
    })
  )[0]?.id || null;
}

2) Replace your current getFirstSceneId(mapData) with this:

function getFirstSceneId(mapData) {
  const alphaFirst = getAlphabeticalFirstSceneId(mapData);
  if (alphaFirst) return alphaFirst;

  const scenes = safeScenes(mapData);
  return Object.keys(scenes)[0] || null;
}

IMPORTANT:
Do NOT use mapData.settings.firstScene here anymore.
That saved value is why Mespack keeps opening first.

3) Keep this rule in StreetViewer:
If URL has ?scene=someId, respect it first.
If there is no ?scene, then use getFirstSceneId(mapData), which is now alphabetical.
